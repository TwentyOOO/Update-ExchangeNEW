📌 الطريقة 1: استخدام CustomBackgroundView (رسومات مخصصة)
═══════════════════════════════════════════════════════════════════

في MainActivity.java، داخل onCreate():

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        // إنشاء الخلفية المخصصة
        FrameLayout container = new FrameLayout(this);

        // 1. إضافة الخلفية
        CustomBackgroundView backgroundView = new CustomBackgroundView(this);
        container.addView(backgroundView, new FrameLayout.LayoutParams(
            FrameLayout.LayoutParams.MATCH_PARENT,
            FrameLayout.LayoutParams.MATCH_PARENT
        ));

        // 2. إضافة المحتوى الأساسي فوق الخلفية
        LinearLayout mainLayout = createMainLayout();  // المحتوى الحالي
        container.addView(mainLayout);

        // 3. تعيين الـ container كمحتوى
        setContentView(container);
    }

الميزات:
- دوائر زخرفية شفافة
- موجات في الأعلى
- شبكة نقاط خفيفة
- خطوط قطرية

📌 أمثلة إضافية - رسم نص على الخلفية
═══════════════════════════════════════════════════════════════════

في CustomBackgroundView.java، أضف:

    private void drawTextWatermark(Canvas canvas, int width, int height) {
        Paint textPaint = new Paint(Paint.ANTI_ALIAS_FLAG);
        textPaint.setColor(Color.parseColor("#10FFFFFF"));
        textPaint.setTextSize(100);
        textPaint.setTextAlign(Paint.Align.CENTER);
        textPaint.setAlpha(30);  // شفافية

        canvas.save();
        canvas.rotate(-45, width / 2, height / 2);  // دوران
        canvas.drawText("ExchangeNEW", width / 2, height / 2, textPaint);
        canvas.restore();
    }

ثم استدعها في onDraw():
    drawTextWatermark(canvas, width, height);

📌 أنماط متقدمة - Pattern/Texture
═══════════════════════════════════════════════════════════════════

    private void drawHexagonPattern(Canvas canvas, int width, int height) {
        Paint hexPaint = new Paint(Paint.ANTI_ALIAS_FLAG);
        hexPaint.setStyle(Paint.Style.STROKE);
        hexPaint.setStrokeWidth(2);
        hexPaint.setColor(Color.parseColor("#10FFFFFF"));

        int size = 50;
        for (int y = 0; y < height; y += size * 1.5) {
            for (int x = 0; x < width; x += size * 2) {
                drawHexagon(canvas, x, y, size, hexPaint);
            }
        }
    }

    private void drawHexagon(Canvas canvas, float x, float y, int size, Paint paint) {
        Path hexPath = new Path();
        for (int i = 0; i < 6; i++) {
            double angle = Math.PI / 3 * i;
            float px = x + size * (float) Math.cos(angle);
            float py = y + size * (float) Math.sin(angle);
            if (i == 0) {
                hexPath.moveTo(px, py);
            } else {
                hexPath.lineTo(px, py);
            }
        }
        hexPath.close();
        canvas.drawPath(hexPath, paint);
    }
